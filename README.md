import pandas as pd
import numpy as np
from sklearn.metrics import confusion_matrix, precision_score, recall_score, f1_score

# Load dataset
df = pd.read_csv('AXISBANK.csv')

# Inspect columns
print("Columns:", df.columns)

# Clean and sort data by Date
df['Date'] = pd.to_datetime(df['Date'])
df.sort_values('Date', inplace=True)

# Compute daily returns
df['Return'] = df['Close'].pct_change()
df.dropna(inplace=True)

# Define anomalies (Actual labels)
# Anomaly if return < -2% (price drop > 2%)
df['Actual'] = (df['Return'] < -0.02).astype(int)

# Predict anomalies (simulate model)
# Here, prediction if return < -1.8%
df['Predicted'] = (df['Return'] < -0.018).astype(int)

# Compute confusion matrix
tn, fp, fn, tp = confusion_matrix(df['Actual'], df['Predicted']).ravel()
# Calculate metrics
precision = precision_score(df['Actual'], df['Predicted'])
recall = recall_score(df['Actual'], df['Predicted'])  # TPR
fpr = fp / (fp + tn)
f1 = f1_score(df['Actual'], df['Predicted'])

# Display results
print("\nConfusion Matrix:")
print(f"TP: {tp}, FP: {fp}, FN: {fn}, TN: {tn}")
print("\nMetrics:")
print(f"Precision: {precision:.2f}")
print(f"Recall (TPR): {recall:.2f}")
print(f"False Positive Rate (FPR): {fpr:.2f}")
print(f"F1 Score: {f1:.2f}")
